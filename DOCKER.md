# Docker Configuration Guide\n\n## Dockerfile Structure\n\nWhen backend is implemented, use this structure:\n\n```dockerfile\n# backend/Dockerfile\nFROM python:3.10-slim\n\nWORKDIR /app\n\n# Install dependencies\nCOPY requirements.txt .\nRUN pip install --no-cache-dir -r requirements.txt\n\n# Copy application\nCOPY . .\n\n# Expose port\nEXPOSE 8000\n\n# Health check\nHEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \\\n  CMD python -c \"import requests; requests.get('http://localhost:8000/health')\"\n\n# Run application\nCMD [\"uvicorn\", \"app.main:app\", \"--host\", \"0.0.0.0\", \"--port\", \"8000\"]\n```\n\n## Docker Compose Setup\n\n```yaml\n# docker-compose.yml\nversion: '3.8'\n\nservices:\n  # Web API\n  web:\n    build: ./backend\n    container_name: bowling-hq-api\n    ports:\n      - \"8000:8000\"\n    environment:\n      - DATABASE_URL=postgresql://user:password@db:5432/bowling_hq\n      - REDIS_URL=redis://redis:6379\n      - SECRET_KEY=${SECRET_KEY}\n      - ENV=development\n    depends_on:\n      - db\n      - redis\n    volumes:\n      - ./backend:/app\n    command: uvicorn app.main:app --host 0.0.0.0 --reload\n    networks:\n      - bowling-network\n\n  # Frontend\n  frontend:\n    build: ./frontend\n    container_name: bowling-hq-web\n    ports:\n      - \"3000:3000\"\n    depends_on:\n      - web\n    environment:\n      - REACT_APP_API_URL=http://localhost:8000\n    volumes:\n      - ./frontend:/app\n    networks:\n      - bowling-network\n\n  # PostgreSQL Database\n  db:\n    image: postgres:13-alpine\n    container_name: bowling-hq-db\n    environment:\n      - POSTGRES_USER=user\n      - POSTGRES_PASSWORD=password\n      - POSTGRES_DB=bowling_hq\n    ports:\n      - \"5432:5432\"\n    volumes:\n      - postgres_data:/var/lib/postgresql/data\n    healthcheck:\n      test: [\"CMD-SHELL\", \"pg_isready -U user\"]\n      interval: 10s\n      timeout: 5s\n      retries: 5\n    networks:\n      - bowling-network\n\n  # Redis Cache\n  redis:\n    image: redis:7-alpine\n    container_name: bowling-hq-redis\n    ports:\n      - \"6379:6379\"\n    volumes:\n      - redis_data:/data\n    healthcheck:\n      test: [\"CMD\", \"redis-cli\", \"ping\"]\n      interval: 10s\n      timeout: 5s\n      retries: 5\n    networks:\n      - bowling-network\n\n  # PgAdmin (Development)\n  pgadmin:\n    image: dpage/pgadmin4:latest\n    container_name: bowling-hq-pgadmin\n    environment:\n      - PGADMIN_DEFAULT_EMAIL=admin@bowling-hq.com\n      - PGADMIN_DEFAULT_PASSWORD=admin\n    ports:\n      - \"5050:80\"\n    depends_on:\n      - db\n    networks:\n      - bowling-network\n\nvolumes:\n  postgres_data:\n  redis_data:\n\nnetworks:\n  bowling-network:\n    driver: bridge\n```\n\n## Common Commands\n\n```bash\n# Build all images\ndocker-compose build\n\n# Start all services\ndocker-compose up -d\n\n# Stop all services\ndocker-compose down\n\n# View logs\ndocker-compose logs -f\n\n# View specific service logs\ndocker-compose logs -f web\n\n# Run migrations\ndocker-compose exec web alembic upgrade head\n\n# Access database CLI\ndocker-compose exec db psql -U user -d bowling_hq\n\n# Access Redis CLI\ndocker-compose exec redis redis-cli\n\n# Rebuild and restart\ndocker-compose up -d --build\n\n# Remove all containers and volumes\ndocker-compose down -v\n```\n\n## Environment Variables\n\n### Development (.env)\n\n```bash\n# Database\nDATABASE_URL=postgresql://user:password@localhost:5432/bowling_hq\n\n# Redis\nREDIS_URL=redis://localhost:6379\n\n# API\nAPI_URL=http://localhost:8000\nFRONTEND_URL=http://localhost:3000\n\n# Security\nSECRET_KEY=dev-secret-key-change-in-production\nJWT_SECRET=dev-jwt-secret-change-in-production\nCORS_ORIGINS=http://localhost:3000\n\n# Environment\nENV=development\nDEBUG=True\n\n# Services\nS3_BUCKET=bowling-hq-dev\nAWS_REGION=us-east-1\n```\n\n## Troubleshooting\n\n### Containers won't start\n\n```bash\n# Check logs\ndocker-compose logs\n\n# Rebuild images\ndocker-compose down\ndocker-compose build --no-cache\ndocker-compose up\n```\n\n### Database connection failed\n\n```bash\n# Check database is running\ndocker-compose ps db\n\n# Check database logs\ndocker-compose logs db\n\n# Reset database\ndocker-compose down -v\ndocker-compose up db\n```\n\n### Port already in use\n\n```bash\n# Find process using port\nlsof -i :8000\n\n# Kill process\nkill -9 <PID>\n\n# Or use different port in docker-compose.yml\nports:\n  - \"8001:8000\"\n```\n\n---\n\n**This is a planned Docker configuration.
